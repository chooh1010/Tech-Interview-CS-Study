class MyStack<T> {
    private Queue<T> q1 = new ArrayDeque<>();
    private Queue<T> q2 = new ArrayDeque<>();

    void push(T x) {
        q2.offer(x);
        while (!q1.isEmpty()) {
            q2.offer(q1.poll());
        }
        Queue<T> temp = q1;
        q1 = q2;
        q2 = temp;
    }

    T pop() {
        return q1.poll();
    }
}
